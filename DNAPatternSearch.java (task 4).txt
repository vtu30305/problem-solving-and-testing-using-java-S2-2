public class DNAPatternSearch {

    // KMP Pattern Search
    static void search(String dna, String pattern) {
        int[] lps = new int[pattern.length()];
        int j = 0;

        // Build LPS array
        for (int i = 1; i < pattern.length();) {
            if (pattern.charAt(i) == pattern.charAt(j))
                lps[i++] = ++j;
            else if (j > 0)
                j = lps[j - 1];
            else
                lps[i++] = 0;
        }

        // Search pattern
        int i = 0;
        while (i < dna.length()) {
            if (dna.charAt(i) == pattern.charAt(j)) {
                i++;
                j++;
            }

            if (j == pattern.length()) {
                System.out.println("Pattern found at position "
                                   + (i - j));
                j = lps[j - 1];
            } else if (i < dna.length()
                    && dna.charAt(i) != pattern.charAt(j)) {
                if (j > 0)
                    j = lps[j - 1];
                else
                    i++;
            }
        }
    }

    public static void main(String[] args) {

        String dna = "ATGCGATGACCTGATG";
        String pattern = "ATG";

        System.out.println("DNA Sequence: " + dna);
        System.out.println("Pattern: " + pattern);

        search(dna, pattern);
    }
}
